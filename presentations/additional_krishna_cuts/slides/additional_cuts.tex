\begin{frame}{$\Delta t$ cut for hadrons}

    \begin{enumerate}
        \item The $\Delta t$ distribution for hadron candidates is divided
        into momentum bins.
    
        \item Each distribution is fitted with a Gaussian to determine
        the mean and $\sigma$ values.
    
        \item A polynomial is fitted to the values
        $\mathrm{mean} \pm 3.5\sigma$ from each momentum bin.
    
        \item Particles within the polynomial boundaries are selected
        for further analysis.
    
        \item The procedure is applied separately to hadrons in the FD and CD.
    \end{enumerate}
    
    \vspace{0.2cm}
    
    %\textcolor{myblue}{\textbf{Time-of-flight difference}}
    
    \[
        \Delta t =
        t_{\mathrm{vx}}^e
        - t_{\mathrm{vx}}^h
    \]
    
    \[
        t_{\mathrm{vx}}^e =
        t_{\mathrm{SC}}^e - \frac{l_{\mathrm{SC}}^e}{c},
        \qquad
        t_{\mathrm{vx}}^h =
        t_{\mathrm{SC}}^h - \frac{l_{\mathrm{SC}}^h}{\beta c},
        \qquad
        \beta = \frac{pc}{E}
    \]
    
    \vspace{-0.1cm}
    
    \begin{center}
    \small
    $l_{\mathrm{SC}}^h$ — hadron path length to SC,
    $t_{\mathrm{SC}}^h$ — hadron time measured in SC

    $l_{\mathrm{SC}}^e$ — electron path length to SC,
    $t_{\mathrm{SC}}^e$ — electron time measured in SC
    \end{center}
    
\end{frame}

\begin{frame}{PCal fiducial cut for electron}

    \textcolor{myblue}{\textbf{Procedure}}
    
    \begin{enumerate}
        \item The $y'$ distribution for electrons is divided into $x'$ bins.
    
        \item The boundaries are determined so that 10\% of the events
        lie outside the boundaries.
    
        \item A polynomial is fitted to the values of the cuts from
        each $x'$ bin.
    
        \item The inner circular cut is determined from the comparison
        with simulated data.
    
        \item Particles within the polynomial boundaries are selected
        for further analysis.

        \item The procedure is applied separately to electrons in each sector of the calorimeter.
    \end{enumerate}
    
    \vspace{0.1cm}
    
    \textcolor{myblue}{\textbf{Coordinate transformation}}
    
    \[
    \begin{aligned}
    x' &= x\cos\left(
        \frac{\pi}{180}(-60\,\mathrm{sec})
        \right)
        - y\sin\left(
        \frac{\pi}{180}(-60\,\mathrm{sec})
        \right),
    \\[0.1cm]
    y' &= x\sin\left(
        \frac{\pi}{180}(-60\,\mathrm{sec})
        \right)
        + y\cos\left(
        \frac{\pi}{180}(-60\,\mathrm{sec})
        \right).
    \end{aligned}
    \]

\end{frame}
