\begin{frame}{Missing energy cut procedure}

    \begin{enumerate}
        \item Generate double pion electroproduction events using
        the twopeg generator.
    
        \item Apply the same standard RGK cuts as for the CLAS12 data.
    
        \item Build distributions of the missing $\pi^+$ absolute
        3-momentum in the COM frame for each bin of $Q^2$ and $W$.
    
        \item For each bin, determine the cut which retains
        $80\%$ of the pions.

        \item Calculate missing energy assuming that $\pi^+$ is on-shell: $E = \sqrt{p^2+m^2}$
    
        \item Apply these cuts to the CLAS12 data.
    \end{enumerate}

\end{frame}
