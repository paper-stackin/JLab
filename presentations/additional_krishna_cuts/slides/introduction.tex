\begin{frame}{Analysis objectives}

    \textcolor{myblue}{\textbf{\large Final goal}}
    
    \begin{itemize}
        \item Obtain nine one-fold differential charge double pion electroproduction
        cross-sections in each bin of $W$ and $Q^2$ covered by the measurements
        in the topology with $\pi^+$ missing.
    \end{itemize}
    
    \vspace{0.5cm}
    
    \textcolor{myblue}{\textbf{\large The current step}}
    
    \begin{itemize}
        \item Evaluation of physical and apparatus backgrounds in exclusive charge
        double pion electroproduction in the topology with $\pi^+$ missing
        in 5D space at any given bin of $W$ and $Q^2$.
    
        \item Implementation of the correlation $ME_{COM}$ cut for cleaning
        the exclusive $\pi^+\pi^-\!p$ event sample.
\end{itemize}

\end{frame}
